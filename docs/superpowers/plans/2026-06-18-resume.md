# 白盒扫描 Resume（断点续传）实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让中断的白盒扫描能用同一 workspace 名重跑、从断点续扫(跳过已完成的 agent),而不是从头开始。

**Architecture:** resume 的本质是"开机重建状态":在 `start_workflow` 前,用 `ResumeStateBuilder` 从磁盘(git log 的 `deliverable:` commit + session.json 的 `metrics.agents[*].status` + 产出物文件)对账重建 `completed_agents`,通过 `PipelineInput.completed_agents` 灌进 workflow,激活项目里现成的"跳过已完成 agent"守卫代码(此前因 `completed_agents` 恒空而空转)。

**Tech Stack:** Python 3 / dataclasses / `subprocess` 调 git / Temporalio(`@workflow.run`) / pytest(asyncio_mode=auto) / click(CLI)。

**Spec:** `docs/superpowers/specs/2026-06-18-resume-design.md`

## Global Constraints

(摘自 spec,所有 task 隐含遵守)
- **agent 粒度**:已完成(有 `deliverable:` commit)的 agent 绝不重跑;不做 agent 内部续传。
- **全量重建**:resume 不依赖 Temporal workflow history,纯靠磁盘三件套。
- **绝不静默退化**:`G ∧ ¬F`(git 有 deliverable commit 但文件丢失)等异常 → abort + 明确错误,**不**悄悄全新扫。
- **completed_agents 用 `AgentName.value` 字符串**(如 `"pre-recon"`),不是枚举名。
- **测试不碰真 Temporal**:单元测试离线(测逻辑片段/mock);E2E 留作人工冒烟。
- **跑测试**:`uv run pytest packages/whitebox/tests/<file>::<test> -v`(asyncio_mode=auto,async 测试无需 `@pytest.mark.asyncio`)。**不要**跑全量(会 hang 在 Temporal 慢测试)。
- **commit 惯例**:`feat(resume): ...` / `test(resume): ...`。
- **deliverables 位置**:`<repo_path>/<deliverables_subdir>`(默认 `<repo>/.shannon/deliverables`),flat 结构;git 操作针对 `repo_path` 所在仓库。
- **commit message 约定**(parse 时依赖,固定):`deliverable: {agent}`、`checkpoint: before {agent} (attempt {n})`。

```python
"""Resume (checkpoint) reconstruction for whitebox scans.

Rebuilds `completed_agents` from disk — git log + session.json + deliverable
files — so an interrupted scan can continue instead of restarting from scratch.
See docs/superpowers/specs/2026-06-18-resume-design.md.
"""
from __future__ import annotations

import subprocess
from dataclasses import dataclass, field
from pathlib import Path

from shannon_core.models.agents import AgentName
from shannon_core.utils.paths import get_default_deliverables_subdir

DELIVERABLE_PREFIX = "deliverable: "
CHECKPOINT_PREFIX = "checkpoint: before "

# Deterministic execution order of whitebox agents that produce deliverables.
# Used to (a) order completed_agents and (b) compute the next agent to run.
WHITEBOX_AGENT_ORDER: list[str] = [
    AgentName.PRE_RECON.value,
    AgentName.RECON.value,
    AgentName.INJECTION_VULN.value,
    AgentName.XSS_VULN.value,
    AgentName.AUTH_VULN.value,
    AgentName.SSRF_VULN.value,
    AgentName.AUTHZ_VULN.value,
]


@dataclass
class ResumeState:
    completed_agents: list[str] = field(default_factory=list)
    interrupted_agent: str | None = None
    base_commit: str | None = None
    warnings: list[str] = field(default_factory=list)
    resume_attempt_number: int = 0


@dataclass
class ResumeStateBuilder:
    repo_path: str
    deliverables_subdir: str = field(default_factory=get_default_deliverables_subdir)
    workspace_name: str | None = None
    workspaces_dir: Path | None = None

    def _run_git_log(self) -> list[tuple[str, str]]:
        """Return list of (commit_hash, message) from newest to oldest."""
        result = subprocess.run(
            ["git", "-C", self.repo_path, "log", "--format=%H%x09%s"],
            capture_output=True, text=True, check=True,
        )
        entries: list[tuple[str, str]] = []
        for line in result.stdout.splitlines():
            if not line.strip():
                continue
            commit_hash, _, message = line.partition("\t")
            entries.append((commit_hash, message))
        return entries

    def _parse_git_completed(self) -> set[str]:
        """Agents that have a `deliverable: {agent}` commit in git log."""
        completed: set[str] = set()
        for _hash, message in self._run_git_log():
            if message.startswith(DELIVERABLE_PREFIX):
                completed.add(message[len(DELIVERABLE_PREFIX):].strip())
        return completed
```

```python
    # --- session.json ---
    def _session_file(self) -> Path | None:
        if not self.workspace_name or not self.workspaces_dir:
            return None
        return Path(self.workspaces_dir) / self.workspace_name / "session.json"

    def _read_session_completed(self) -> set[str]:
        """Agents with metrics.agents[*].status == 'success' in session.json."""
        path = self._session_file()
        if not path or not path.exists():
            return set()
        try:
            data = json.loads(path.read_text(encoding="utf-8"))
        except (json.JSONDecodeError, OSError):
            return set()
        agents = (data.get("metrics") or {}).get("agents") or {}
        return {name for name, info in agents.items() if info.get("status") == "success"}

    def _read_resume_attempt(self) -> int:
        """Number of prior resume attempts recorded in session.json (len of resumeAttempts list)."""
        path = self._session_file()
        if not path or not path.exists():
            return 0
        try:
            data = json.loads(path.read_text(encoding="utf-8"))
        except (json.JSONDecodeError, OSError):
            return 0
        attempts = data.get("resumeAttempts")
        return len(attempts) if isinstance(attempts, list) else 0

    # --- deliverable file check ---
    def _deliverable_path(self, agent_value: str) -> Path | None:
        from shannon_core.models.agents import AGENTS
        try:
            agent_enum = AgentName(agent_value)
        except ValueError:
            return None
        filename = AGENTS[agent_enum].deliverable_filename
        if filename is None:
            return None
        return Path(self.repo_path) / self.deliverables_subdir / filename

    def _check_deliverable_exists(self, agent_value: str) -> bool:
        path = self._deliverable_path(agent_value)
        return path is not None and path.exists()

    # --- reconcile (decision table; git authoritative) ---
    def _reconcile(self, git_completed: set[str], json_completed: set[str]) -> tuple[list[str], list[str]]:
        completed: list[str] = []
        warnings: list[str] = []
        for agent in WHITEBOX_AGENT_ORDER:
            in_git = agent in git_completed
            in_json = agent in json_completed
            if not in_git:
                if in_json:
                    warnings.append(
                        f"session.json marks '{agent}' as success but no deliverable commit found; will re-run"
                    )
                continue
            if not self._check_deliverable_exists(agent):
                raise ResumeAbortedError(
                    f"git has a deliverable commit for '{agent}' but the deliverable file is missing on disk; "
                    f"aborting resume to avoid silent data loss (manual intervention required)"
                )
            if not in_json:
                warnings.append(
                    f"git has a deliverable commit for '{agent}' but session.json has no success record; "
                    f"counting as completed (git authoritative)"
                )
            completed.append(agent)
        return completed, warnings
```

```python
    # --- interrupted agent + base commit ---
    def _next_agent_after(self, completed: set[str]) -> str | None:
        for agent in WHITEBOX_AGENT_ORDER:
            if agent not in completed:
                return agent
        return None

    def _locate_interrupted_and_base(self) -> tuple[str | None, str | None]:
        """Newest dangling `checkpoint: before X` → X is interrupted (base = that commit);
        otherwise newest `deliverable: X` → next agent to run (base = that commit)."""
        delivered = self._parse_git_completed()
        for commit_hash, message in self._run_git_log():
            if message.startswith(CHECKPOINT_PREFIX):
                rest = message[len(CHECKPOINT_PREFIX):]
                agent = rest.split(" (attempt")[0].strip()
                if agent and agent not in delivered:
                    return agent, commit_hash
            if message.startswith(DELIVERABLE_PREFIX):
                agent = message[len(DELIVERABLE_PREFIX):].strip()
                return self._next_agent_after(delivered), commit_hash
        return None, None

    # --- public entry point ---
    def load_resume_state(self) -> ResumeState:
        git_completed = self._parse_git_completed()
```
